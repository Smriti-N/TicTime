// Game state and event wiring const state = { board: createEmptyBoard(), currentPlayer: 'X', gameOver: false, scores: { X: 0, O: 0, draws: 0 }, theme: 'light', }; function handleCellClick(index) { if (state.gameOver || state.board[index] !== '') return; state.board[index] = state.currentPlayer; renderBoard(state.board); const result = getGameResult(state.board); if (result.status === 'win') { state.gameOver = true; state.scores[result.winner]++; highlightWinningCells(result.line); setStatus(`Player ${result.winner} wins!`, result.winner); setBoardLocked(true); renderScores(state.scores); launchConfetti(); } else if (result.status === 'draw') { state.gameOver = true; state.scores.draws++; setStatus("It's a draw!", ''); setBoardLocked(true); renderScores(state.scores); } else { state.currentPlayer = nextPlayer(state.currentPlayer); setStatus(`Player ${state.currentPlayer}'s turn`, state.currentPlayer); } } // Start a new game function startNewGame() { state.board = createEmptyBoard(); state.currentPlayer = 'X'; state.gameOver = false; renderBoard(state.board); setBoardLocked(false); setStatus("Player X's turn", 'X'); } // Reset all scores function resetScores() { state.scores = { X: 0, O: 0, draws: 0 }; renderScores(state.scores); startNewGame(); } // Toggle light/dark theme function toggleTheme() { state.theme = state.theme === 'light' ? 'dark' : 'light'; setTheme(state.theme); } // Add footer function createFooter() { const footer = document.createElement('footer'); footer.textContent = '© 2026 Tic Tac Toe | Play • Compete • Enjoy'; footer.style.textAlign = 'center'; footer.style.padding = '20px'; footer.style.marginTop = '30px'; footer.style.fontSize = '14px'; footer.style.opacity = '0.7'; document.body.appendChild(footer); } // Initialize game buildBoard(handleCellClick); renderBoard(state.board); renderScores(state.scores); setStatus("Player X's turn", 'X'); document.getElementById('new-game-btn') .addEventListener('click', startNewGame); document.getElementById('theme-toggle') .addEventListener('click', toggleTheme); document.getElementById('reset-scores-btn') .addEventListener('click', resetScores); // Add footer to page createFooter();
